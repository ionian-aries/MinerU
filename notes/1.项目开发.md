# 1) 在项目目录创建虚拟环境
uv venv --python 3.12
# 2) 激活虚拟环境
source .venv/bin/activate

## 提高超时 + 降低并发，重试（可选）
export UV_HTTP_TIMEOUT=180
export UV_CONCURRENT_DOWNLOADS=2

# 3) 再安装（开发模式 + 全量依赖）
uv pip install -e ".[all]"
uv pip install -e ".[all]" -i https://mirrors.aliyun.com/pypi/simple （或者使用国内的源）
然后验证：

mineru --help
